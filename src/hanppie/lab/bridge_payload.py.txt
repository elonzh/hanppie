CONTROL_PORT = __CONTROL_PORT__
TELEMETRY_PORT = __TELEMETRY_PORT__
TELEMETRY_PERIOD = __TELEMETRY_PERIOD__
COMMAND_TIMEOUT = __COMMAND_TIMEOUT__
MAX_CHASSIS_SPEED = 3.5
MAX_CHASSIS_YAW_SPEED = 600.0
MAX_GIMBAL_SPEED = 540.0


def _root_import(name):
    builtins = rm_define.__dict__["__builtins__"]
    importer = builtins["__import__"] if isinstance(builtins, dict) else builtins.__import__
    return importer(name, globals(), locals(), [], 0)


def _safe_call(function, *arguments):
    try:
        result = function(*arguments)
        if result is False:
            return False
        if isinstance(result, dict):
            values = result.get("kw", {})
            if isinstance(values, dict) and "result" in values:
                return values.get("result") in (0, True)
        return True
    except Exception:
        return False


def _record_call(state, name, function, *arguments):
    try:
        result = function(*arguments)
        state["last_command"] = name
        state["last_command_ok"] = result is not False
        state["last_command_error"] = ""
        return state["last_command_ok"]
    except Exception as error:
        state["last_command"] = name
        state["last_command_ok"] = False
        state["last_command_error"] = str(error)
        return False


def _clamp(value, minimum, maximum):
    try:
        number = float(value)
    except Exception:
        number = 0.0
    if number < minimum:
        return minimum
    if number > maximum:
        return maximum
    return number


def _has_field(command, names):
    for name in names:
        if name in command:
            return True
    return False


def _stop_motion():
    _safe_call(chassis_ctrl.stop)
    _safe_call(gimbal_ctrl.stop)


def _set_mode(name):
    mode = str(name).lower()
    if mode in ("free", "gimbal_lead"):
        return _safe_call(robot_ctrl.set_mode, rm_define.robot_mode_free)
    if mode in ("gimbal_follow", "chassis_lead"):
        return _safe_call(robot_ctrl.set_mode, rm_define.robot_mode_gimbal_follow)
    if mode in ("chassis_follow", "follow"):
        return _safe_call(robot_ctrl.set_mode, rm_define.robot_mode_chassis_follow)
    return False


def _set_led(command):
    red = int(_clamp(command.get("red", 255), 0, 255))
    green = int(_clamp(command.get("green", 255), 0, 255))
    blue = int(_clamp(command.get("blue", 255), 0, 255))
    component = str(command.get("component", "all")).lower()
    effect_name = str(command.get("effect", "on")).lower()
    if effect_name == "off":
        if component in ("all", "bottom"):
            _safe_call(led_ctrl.turn_off, rm_define.armor_bottom_all)
        if component in ("all", "top", "gimbal"):
            _safe_call(led_ctrl.turn_off, rm_define.armor_top_all)
        return
    effects = {
        "on": rm_define.effect_always_on,
        "solid": rm_define.effect_always_on,
        "breath": rm_define.effect_breath,
        "flash": rm_define.effect_flash,
    }
    effect = effects.get(effect_name, rm_define.effect_always_on)
    if component in ("all", "bottom"):
        _safe_call(
            led_ctrl.set_bottom_led,
            rm_define.armor_bottom_all,
            red,
            green,
            blue,
            effect,
        )
    if component in ("all", "top", "gimbal"):
        _safe_call(
            led_ctrl.set_top_led,
            rm_define.armor_top_all,
            red,
            green,
            blue,
            effect,
        )


def _generic_call(command):
    module = str(command.get("module", "")).lower()
    method = str(command.get("method", "")).lower()
    params = command.get("params", {})
    if not isinstance(params, dict):
        return False
    if module == "system":
        return method == "set_telemetry"
    if module == "chassis":
        if method == "stop":
            return _safe_call(chassis_ctrl.stop)
        if method == "set_wheel_speed":
            return _safe_call(
                chassis_ctrl.set_wheel_speed,
                int(params.get("w1", 0)),
                int(params.get("w2", 0)),
                int(params.get("w3", 0)),
                int(params.get("w4", 0)),
            )
        if method == "set_pwm_value":
            return _safe_call(
                chassis_ctrl.set_pwm_value,
                int(params.get("pwm", 0)),
                float(params.get("value", 0)),
            )
    if module == "gimbal":
        if method == "stop":
            return _safe_call(gimbal_ctrl.stop)
        if method == "recenter":
            return _safe_call(gimbal_ctrl.recenter)
        if method == "suspend":
            return _safe_call(gimbal_ctrl.suspend)
        if method == "resume":
            return _safe_call(gimbal_ctrl.resume)
        if method == "angle_ctrl":
            return _safe_call(
                gimbal_ctrl.angle_ctrl,
                float(params.get("yaw", 0)),
                float(params.get("pitch", 0)),
            )
    if module == "armor" and method == "set_hit_sensitivity":
        return _safe_call(armor_ctrl.set_hit_sensitivity, int(params.get("sensitivity", 5)))
    if module == "blaster":
        effect = str(params.get("effect", "on")).lower()
        enabled = effect not in ("off", "0", "false")
        if method == "set_led":
            return _safe_call(led_ctrl.gun_led_on if enabled else led_ctrl.gun_led_off)
        if method == "set_fire_led":
            return _safe_call(led_ctrl.fire_led_on if enabled else led_ctrl.fire_led_off)
        if method == "reset_led":
            fire_off = _safe_call(led_ctrl.fire_led_off)
            gun_off = _safe_call(led_ctrl.gun_led_off)
            return fire_off and gun_off
    if module == "media":
        if method == "play_sound":
            return _safe_call(media_ctrl.play_sound, params.get("sound", 0))
        if method == "capture":
            return _safe_call(media_ctrl.capture)
    return False


def _apply_command(command, state, now):
    session_id = int(command.get("session_id", 0) or 0)
    sequence = int(command.get("command_seq", 0) or 0)
    if not session_id or not sequence:
        return
    if session_id != state["session_id"]:
        _stop_motion()
        state["session_id"] = session_id
        state["last_sequence"] = 0
        state["armed"] = False
        state["chassis_active"] = False
        state["gimbal_active"] = False
        state["motion_active"] = False
    if sequence <= state["last_sequence"]:
        return
    state["last_sequence"] = sequence

    module = str(command.get("module", "")).lower()
    method = str(command.get("method", "")).lower()
    if module == "system" and method == "arm":
        state["armed"] = True
        state["last_command"] = "system.arm"
        state["last_command_ok"] = True
        state["last_command_error"] = ""
        return
    if module == "system" and method == "disarm":
        state["armed"] = False
        state["chassis_active"] = False
        state["gimbal_active"] = False
        state["motion_active"] = False
        _stop_motion()
        state["last_command"] = "system.disarm"
        state["last_command_ok"] = True
        state["last_command_error"] = ""
        return
    if command.get("stop", False):
        state["chassis_active"] = False
        state["gimbal_active"] = False
        state["motion_active"] = False
        _stop_motion()
        state["last_command"] = "system.stop"
        state["last_command_ok"] = True
        state["last_command_error"] = ""
        return
    if command.get("led", False):
        _set_led(command)
        state["last_command"] = "led.set"
        state["last_command_ok"] = True
        state["last_command_error"] = ""
    if module and method and module not in ("chassis", "gimbal"):
        state["last_command"] = module + "." + method
        state["last_command_ok"] = _generic_call(command)
        state["last_command_error"] = "" if state["last_command_ok"] else "rejected"
    if not state["armed"]:
        return
    if "mode" in command:
        state["last_command"] = "robot.set_mode"
        state["last_command_ok"] = _set_mode(command.get("mode"))
        state["last_command_error"] = "" if state["last_command_ok"] else "rejected"
    if module in ("chassis", "gimbal") and method:
        state["last_command"] = module + "." + method
        state["last_command_ok"] = _generic_call(command)
        state["last_command_error"] = "" if state["last_command_ok"] else "rejected"
        if module == "chassis":
            state["chassis_active"] = False
        else:
            state["gimbal_active"] = False
        state["motion_active"] = state["chassis_active"] or state["gimbal_active"]
    if command.get("fire", False):
        fire_type = str(command.get("fire_type", "infrared")).lower()
        led_ok = _safe_call(led_ctrl.fire_led_on)
        sound_ok = _safe_call(media_ctrl.play_sound, rm_define.media_sound_shoot)
        if fire_type in ("infrared", "ir", "led"):
            fire_name = "blaster.fire_ir"
            fire_ok = _safe_call(ir_blaster_ctrl.fire_once)
        else:
            fire_name = "blaster.fire_gel"
            fire_ok = _safe_call(gun_ctrl.fire_once)
        state["last_fire_led_ok"] = led_ok
        state["last_fire_sound_ok"] = sound_ok
        state["last_fire_actuator_ok"] = fire_ok
        state["fire_led_off_time"] = now + 0.4
        state["last_command"] = fire_name
        state["last_command_ok"] = led_ok and sound_ok and fire_ok
        failures = []
        if not led_ok:
            failures.append("muzzle-led")
        if not sound_ok:
            failures.append("shoot-sound")
        if not fire_ok:
            failures.append("actuator")
        state["last_command_error"] = ",".join(failures)

    chassis_fields = ("x", "y", "z")
    gimbal_fields = ("gimbal_pitch", "gimbal_yaw")
    has_chassis = _has_field(command, chassis_fields)
    has_gimbal = _has_field(command, gimbal_fields)
    if has_chassis:
        state["chassis_active"] = _record_call(
            state,
            "chassis.move_with_speed",
            chassis_ctrl.move_with_speed,
            _clamp(command.get("x", 0), -MAX_CHASSIS_SPEED, MAX_CHASSIS_SPEED),
            _clamp(command.get("y", 0), -MAX_CHASSIS_SPEED, MAX_CHASSIS_SPEED),
            _clamp(command.get("z", 0), -MAX_CHASSIS_YAW_SPEED, MAX_CHASSIS_YAW_SPEED),
        )
    if has_gimbal:
        state["gimbal_active"] = _record_call(
            state,
            "gimbal.rotate_with_speed",
            gimbal_ctrl.rotate_with_speed,
            int(_clamp(command.get("gimbal_yaw", 0), -MAX_GIMBAL_SPEED, MAX_GIMBAL_SPEED)),
            int(_clamp(command.get("gimbal_pitch", 0), -MAX_GIMBAL_SPEED, MAX_GIMBAL_SPEED)),
        )
    if has_chassis or has_gimbal:
        state["last_motion"] = now
        state["motion_active"] = state["chassis_active"] or state["gimbal_active"]


def _value(function, argument):
    try:
        return float(function(argument))
    except Exception:
        return None


def _telemetry(state, now):
    return {
        "type": "telemetry",
        "sequence": state["telemetry_sequence"],
        "time_ms": int(now * 1000),
        "session_id": state["session_id"],
        "rx_command_seq": state["last_sequence"],
        "armed": state["armed"],
        "chassis_active": state["chassis_active"],
        "gimbal_active": state["gimbal_active"],
        "motion_active": state["motion_active"],
        "last_command": state["last_command"],
        "last_command_ok": state["last_command_ok"],
        "last_command_error": state["last_command_error"],
        "last_fire_led_ok": state["last_fire_led_ok"],
        "last_fire_sound_ok": state["last_fire_sound_ok"],
        "last_fire_actuator_ok": state["last_fire_actuator_ok"],
        "x": _value(chassis_ctrl.get_position_based_power_on, rm_define.chassis_forward),
        "y": _value(chassis_ctrl.get_position_based_power_on, rm_define.chassis_translation),
        "yaw": _value(chassis_ctrl.get_attitude, rm_define.chassis_yaw),
        "gimbal_yaw": _value(gimbal_ctrl.get_axis_angle, rm_define.gimbal_axis_yaw),
        "gimbal_pitch": _value(gimbal_ctrl.get_axis_angle, rm_define.gimbal_axis_pitch),
    }


def start():
    time_module = _root_import("time")
    socket_module = _root_import("socket")
    select_module = _root_import("select")
    json_module = _root_import("json")
    _set_mode("free")
    _stop_motion()

    receiver = socket_module.socket(socket_module.AF_INET, socket_module.SOCK_DGRAM)
    receiver.setsockopt(socket_module.SOL_SOCKET, socket_module.SO_REUSEADDR, 1)
    receiver.bind(("0.0.0.0", CONTROL_PORT))
    receiver.setblocking(False)
    sender = socket_module.socket(socket_module.AF_INET, socket_module.SOCK_DGRAM)
    sender.setsockopt(socket_module.SOL_SOCKET, socket_module.SO_BROADCAST, 1)
    state = {
        "target": ("255.255.255.255", TELEMETRY_PORT),
        "session_id": 0,
        "last_sequence": 0,
        "telemetry_sequence": 0,
        "armed": False,
        "chassis_active": False,
        "gimbal_active": False,
        "motion_active": False,
        "last_motion": 0.0,
        "last_command": "",
        "last_command_ok": True,
        "last_command_error": "",
        "last_fire_led_ok": None,
        "last_fire_sound_ok": None,
        "last_fire_actuator_ok": None,
        "fire_led_off_time": 0.0,
    }
    next_telemetry = time_module.time()
    while True:
        readable, _, _ = select_module.select([receiver], [], [], 0.01)
        for ready in readable:
            try:
                payload, source = ready.recvfrom(4096)
                command = json_module.loads(payload.decode("utf-8"))
                if isinstance(command, dict):
                    state["target"] = (source[0], TELEMETRY_PORT)
                    _apply_command(command, state, time_module.time())
            except Exception:
                pass
        now = time_module.time()
        if state["motion_active"] and now - state["last_motion"] > COMMAND_TIMEOUT:
            _stop_motion()
            state["chassis_active"] = False
            state["gimbal_active"] = False
            state["motion_active"] = False
        if state["fire_led_off_time"] and now >= state["fire_led_off_time"]:
            _safe_call(led_ctrl.fire_led_off)
            state["fire_led_off_time"] = 0.0
        if now >= next_telemetry:
            packet = json_module.dumps(_telemetry(state, now), separators=(",", ":"))
            try:
                sender.sendto(packet.encode("utf-8"), state["target"])
            except Exception:
                pass
            state["telemetry_sequence"] = state["telemetry_sequence"] + 1
            next_telemetry = now + TELEMETRY_PERIOD
